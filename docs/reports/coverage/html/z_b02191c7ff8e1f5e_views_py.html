21def _dates(request): 

22 """date_from / date_to from the query string; anything unreadable is ignored.""" 

23 out = [] 

24 for name in ("date_from", "date_to"): 

25 try: 

26 out.append(dt.date.fromisoformat(request.GET.get(name, ""))) 

27 except ValueError: 

28 out.append(None) 

29 return out

32def _show(request, report): 

33 if request.GET.get("format") == "csv": 

34 return export.as_csv(report) 

35 if request.GET.get("format") == "pdf": 

36 return export.as_pdf(report, generated_by=request.user.username) 

37 date_from, date_to = _dates(request) 

38 page = Paginator(_flagged_rows(report), PER_PAGE).get_page(request.GET.get("page")) 

39 return render(

43 "threshold": get_threshold()},

47def _flagged_rows(report): 

48 """Each row as ([(cell, below_threshold), ...], student_id), so low percentages can be highlighted.""" 

49 threshold = get_threshold() 

50 rows = [] 

51 for cells, student_id in zip(report.rows, report.links, strict=True): 

52 flagged = [] 

53 for i, cell in enumerate(cells): 

54 low = False 

55 if i in report.percent_columns and isinstance(cell, str) and cell.endswith("%"): 

56 low = is_below(float(cell[:-1]), threshold) 

57 flagged.append((cell, low)) 

58 rows.append((flagged, student_id)) 

59 return rows
